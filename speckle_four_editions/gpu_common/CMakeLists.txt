cmake_minimum_required(VERSION 3.16)
project(speckle_native VERSION 0.2.0 LANGUAGES CXX CUDA)
set(SPECKLE_ARCH "89" CACHE STRING "CUDA SM architecture:89 desktop Ada,87 Orin")
set(SPECKLE_PLATFORM "desktop" CACHE STRING "desktop or jetson")
if(NOT SPECKLE_ARCH MATCHES "^(87|89)$" OR NOT SPECKLE_PLATFORM MATCHES "^(desktop|jetson)$")
  message(FATAL_ERROR "Supported presets: desktop SM89 or jetson SM87")
endif()
find_package(CUDA REQUIRED)
add_library(speckle_native SHARED native/edge_kernels.cu src/session.cu)
target_include_directories(speckle_native PUBLIC include PRIVATE ${CUDA_INCLUDE_DIRS})
target_compile_options(speckle_native PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;--std=c++17;--fmad=false;-arch=sm_${SPECKLE_ARCH}>")
target_link_libraries(speckle_native PRIVATE ${CUDA_CUFFT_LIBRARIES} ${CUDA_CUDART_LIBRARY})
set_target_properties(speckle_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
add_executable(speckle_gpu src/main.cpp)
target_compile_features(speckle_gpu PRIVATE cxx_std_17)
target_compile_options(speckle_gpu PRIVATE -O3)
target_compile_definitions(speckle_gpu PRIVATE SPECKLE_PLATFORM="${SPECKLE_PLATFORM}")
target_link_libraries(speckle_gpu PRIVATE speckle_native)
set_target_properties(speckle_gpu PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS speckle_native speckle_gpu LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/speckle_native.h DESTINATION include)
